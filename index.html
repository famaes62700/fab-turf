async function loadLivePmu() { const dateInput = document.getElementById('raceDate').value; const statusText = document.getElementById('statusText'); statusText.innerHTML = 'Chargement du programme PMU...'; const parts = dateInput.split('-'); const ddmmyyyy = `${parts[2]}${parts[1]}${parts[0]}`; try { const response = await fetch(`/api/courses?date=${ddmmyyyy}`); if (!response.ok) { throw new Error("Erreur de connexion au serveur proxy"); } const data = await response.json(); // Gestion flexible des structures de données PMU let reunions = null; if (data && data.programme && data.programme.reunions) { reunions = data.programme.reunions; } else if (data && data.reunions) { reunions = data.reunions; } if (reunions && reunions.length > 0) { liveMeetingsData = reunions; populateMeetingsSelect(liveMeetingsData); statusText.innerHTML = `Programme chargé (${liveMeetingsData.length} réunions)`; } else { throw new Error("Aucune réunion disponible pour cette date."); } } catch (err) { console.error(err); statusText.innerHTML = `Erreur de liaison réseau avec le PMU`; clearDisplay(); } }

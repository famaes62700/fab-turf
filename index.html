async function loadLivePmu() { const dateInput = document.getElementById('raceDate').value; // Format AAAA-MM-JJ const statusText = document.getElementById('statusText'); statusText.innerHTML = 'Chargement des courses...'; const parts = dateInput.split('-'); const ddmmyyyy = `${parts[2]}${parts[1]}${parts[0]}`; try { // On interroge directement VOTRE relais Vercel personnel const response = await fetch(`https://relais-turf.vercel.app/api/courses?date=${ddmmyyyy}`); if (!response.ok) throw new Error("Erreur de communication avec votre relais"); const data = await response.json(); if (data && data.programme && data.programme.reunions) { liveMeetingsData = data.programme.reunions; populateMeetingsSelect(liveMeetingsData); statusText.innerHTML = `Programme chargé (${liveMeetingsData.length} réunions).`; } else { throw new Error("Aucune course disponible pour cette date."); } } catch (err) { console.error(err); statusText.innerHTML = `Erreur : ${err.message}. Essayez la date du jour.`; clearDisplay(); } }
